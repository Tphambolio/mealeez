3) On the client, detect Android lack of SpeechRecognition and fall back to MediaRecorder→/api/stt

Replace your startListening logic with:

const hasSR = !!(window as any).webkitSpeechRecognition || !!(window as any).SpeechRecognition;

async function startListening(){
  setTranscript("");
  if (hasSR) {
    keepListeningRef.current = true;
    recognitionRef.current!.start();
    setIsListening(true);
    return;
  }
  // Fallback: record a short utterance and send to STT
  const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
  const rec = new MediaRecorder(stream, { mimeType: "audio/webm" });
  const chunks: BlobPart[] = [];
  rec.ondataavailable = (e) => chunks.push(e.data);
  rec.onstop = async () => {
    const blob = new Blob(chunks, { type: "audio/webm" });
    const form = new FormData();
    form.append("audio", blob, "speech.webm");
    const r = await fetch("/api/stt", { method: "POST", body: form });
    const { text } = await r.json();
    if (text?.trim()) await handleVoiceCommand(text.trim());
  };
  rec.start();
  setIsListening(true);
  // Stop after a pause window (e.g., 4s); you can wire a press-and-hold UI instead.
  setTimeout(() => { rec.stop(); stream.getTracks().forEach(t=>t.stop()); setIsListening(false); }, 4000);
}

> Why: Android Chrome frequently lacks/cripples SpeechRecognition. MediaRecorder works broadly; the server handles transcription.

5) On the client, pause mic while TTS speaks, auto-resume, and parse the streamed JSON tail

Add:

function extractLastJSONObject(text:string){
  const last = text.lastIndexOf("{");
  for (let i = last; i >= 0; i--) {
    try { return JSON.parse(text.slice(i).trim()); } catch {}
  }
  return null;
}

async function sendStreaming(route:string, payload:any){
  const res = await fetch(route, { method:"POST", headers:{ "Content-Type":"application/json" }, body: JSON.stringify(payload) });
  if (!res.body) throw new Error("no stream");
  const reader = res.body.getReader(); const dec = new TextDecoder(); let full="";
  for(;;){ const {value,done}=await reader.read(); if(done) break; full += dec.decode(value,{stream:true}); }
  return extractLastJSONObject(full);
}

const handleVoiceCommand = async (command:string) => {
  const wasListening = isListening;
  if (wasListening) stopListening();
  const json = await sendStreaming("/api/voice/plan-meal", { transcript: command, clientState: getClientState?.() || {} });
  if (json?.reply) { window.speechSynthesis.cancel(); const u=new SpeechSynthesisUtterance(json.reply); u.onend=()=>{ setIsSpeaking(false); if (wasListening) startListening(); }; setIsSpeaking(true); speechSynthesis.speak(u); }
  if (Array.isArray(json?.actions)) applyActions(json.actions);
};
