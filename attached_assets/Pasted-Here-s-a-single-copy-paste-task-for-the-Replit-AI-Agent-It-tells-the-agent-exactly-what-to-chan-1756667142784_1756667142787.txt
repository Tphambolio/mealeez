```ts
    res.setHeader("Content-Type", "text/plain; charset=utf-8");
    res.setHeader("Cache-Control", "no-cache");
    res.setHeader("Connection", "keep-alive");

    const r = await fetch("https://api.openai.com/v1/chat/completions", {

        stream: true,
        messages: [

          { role: "user", content: JSON.stringify({

              state: req.body.clientState || {}
          })}
        ]
      })
    });

    if (!r.ok || !r.body) {
      const errTxt = await r.text().catch(() => "");
      console.error("OpenAI upstream error:", r.status, errTxt);
      res.status(502).end(`upstream_error:${r.status}`);
      return;
    }

    for await (const chunk of r.body as any) res.write(chunk);
    res.end();
  } catch (e) {
    console.error("plan-meal route error:", e);
    res.status(500).end("route_crash");
  }
});
```

### 3) Client: stream reader + JSON tail extraction

**Add utility** (or inline it) used by the voice component:

```ts
export async function sendStreaming(route: string, payload: any) {
  const res = await fetch(route, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload)
  });
  if (!res.body) throw new Error("no response body");
  const reader = res.body.getReader();
  const dec = new TextDecoder();
  let full = "";
  for (;;) {
    const { value, done } = await reader.read();
    if (done) break;
    full += dec.decode(value, { stream: true });
    // Optional: show typing animation here
  }
  // Extract the last valid JSON object (schema block) from the stream tail
  for (let i = full.lastIndexOf("{"); i >= 0; i--) {
    try { return JSON.parse(full.slice(i).trim()); } catch {}
  }
  throw new Error("no JSON found in stream");
}
```

**Use it** inside the voice handler:


### 4) Android STT fallback (when `SpeechRecognition` is missing)

Add a fallback in the voice component:

```ts
const hasSR = !!(window as any).webkitSpeechRecognition || !!(window as any).SpeechRecognition;

async function startListening(){
  setTranscript("");
  if (hasSR) {
    keepListeningRef.current = true;
    recognitionRef.current!.start();
    setIsListening(true);
    return;
  }
  // Fallback: short-record with MediaRecorder (press-to-talk UX)
  const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
  const rec = new MediaRecorder(stream, { mimeType: "audio/webm" });
  const chunks: BlobPart[] = [];
  rec.ondataavailable = (e) => chunks.push(e.data);
  rec.onstop = async () => {
    const blob = new Blob(chunks, { type: "audio/webm" });
    const form = new FormData(); form.append("audio", blob, "speech.webm");
    const r = await fetch("/api/stt", { method: "POST", body: form });
    const { text } = await r.json();
    if (text?.trim()) await handleVoiceCommand(text.trim());
    stream.getTracks().forEach(t => t.stop());
  };
  rec.start();
  setIsListening(true);
  setTimeout(() => { rec.stop(); setIsListening(false); }, 4000); // basic VAD window
}
```
